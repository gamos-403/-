--  Цвета: Выжившие - ЗЕЛЁНЫЕ, Убийца - КРАСНЫЙ, Генераторы - ЖЁЛТЫЕ
--  Фичи: ESP + Прицел + Fullbright + Плавающая кнопка меню

local Players = game:GetService("Players")
local Lighting = game:GetService("Lighting")
local UserInputService = game:GetService("UserInputService")
local LocalPlayer = Players.LocalPlayer

-- ============================================================
--  ГЛОБАЛЬНЫЕ ПЕРЕМЕННЫЕ
-- ============================================================
local espOn = false
local fullbrightOn = false
local crosshairOn = false
local oldBrightness = Lighting.Brightness
local oldAmbient = Lighting.Ambient
local highlights = {}
local menuOpen = false

-- ============================================================
--  ФУНКЦИЯ ПОИСКА УБИЙЦЫ
-- ============================================================
local function findKiller()
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LocalPlayer then continue end
        local char = plr.Character
        if char then
            for _, tool in ipairs(char:GetChildren()) do
                if tool:IsA("Tool") then
                    local toolName = tool.Name:lower()
                    if toolName:find("knife") or toolName:find("gun") or toolName:find("weapon") or 
                       toolName:find("sword") or toolName:find("blade") or toolName:find("пистолет") or 
                       toolName:find("нож") or toolName:find("меч") then
                        return plr
                    end
                end
            end
        end
    end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LocalPlayer then continue end
        local team = plr.Team
        if team then
            local teamName = team.Name:lower()
            if teamName:find("killer") or teamName:find("murderer") or teamName:find("impostor") or 
               teamName:find("traitor") or teamName:find("убийц") or teamName:find("предатель") then
                return plr
            end
        end
    end
    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LocalPlayer then continue end
        if plr.TeamColor then
            local colorName = plr.TeamColor.Name:lower()
            if colorName:find("red") or colorName:find("красн") then
                return plr
            end
        end
    end
    return nil
end

-- ============================================================
--  ФУНКЦИИ ESP
-- ============================================================
local function clearHighlights()
    for _, obj in ipairs(highlights) do
        if obj and obj.Parent then
            obj:Destroy()
        end
    end
    highlights = {}
end

local function applyESP()
    clearHighlights()
    if not espOn then return end

    local killer = findKiller()

    for _, plr in ipairs(Players:GetPlayers()) do
        if plr == LocalPlayer then continue end
        local char = plr.Character
        if not char then continue end
        local hum = char:FindFirstChild("Humanoid")
        if not hum or hum.Health <= 0 then continue end

        local highlight = Instance.new("Highlight")
        if plr == killer then
            highlight.FillColor = Color3.fromRGB(255, 0, 0) -- КРАСНЫЙ (убийца)
        else
            highlight.FillColor = Color3.fromRGB(0, 255, 0) -- ЗЕЛЁНЫЙ (выжившие)
        end
        highlight.FillTransparency = 0.4
        highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
        highlight.OutlineTransparency = 0.2
        highlight.Adornee = char
        highlight.Parent = char
        table.insert(highlights, highlight)
    end

    for _, obj in ipairs(workspace:GetDescendants()) do
        if obj:IsA("BasePart") then
            local name = obj.Name:lower()
            if name:find("generator") or name:find("gen") or name:find("генератор") or 
               name:find("reactor") or name:find("engine") then
                local highlight = Instance.new("Highlight")
                highlight.FillColor = Color3.fromRGB(255, 255, 0) -- ЖЁЛТЫЙ
                highlight.FillTransparency = 0.3
                highlight.OutlineColor = Color3.fromRGB(255, 255, 255)
                highlight.OutlineTransparency = 0.1
                highlight.Adornee = obj
                highlight.Parent = obj
                table.insert(highlights, highlight)
            end
        end
    end
end

-- ============================================================
--  ФУНКЦИЯ СОЗДАНИЯ GUI
-- ============================================================
local function createGUI()
    local oldGui = LocalPlayer.PlayerGui:FindFirstChild("DistrictOfViolence")
    if oldGui then oldGui:Destroy() end

    local oldCrosshair = LocalPlayer.PlayerGui:FindFirstChild("CrosshairGui")
    if oldCrosshair then oldCrosshair:Destroy() end

    local screenGui = Instance.new("ScreenGui")
    screenGui.Name = "DistrictOfViolence"
    screenGui.ResetOnSpawn = false
    screenGui.Parent = LocalPlayer.PlayerGui

    local toggleButton = Instance.new("ImageButton")
    toggleButton.Size = UDim2.new(0, 60, 0, 60)
    toggleButton.Position = UDim2.new(0.85, -30, 0.85, -30)
    toggleButton.BackgroundColor3 = Color3.fromRGB(40, 40, 40)
    toggleButton.BorderSizePixel = 0
    toggleButton.Image = "rbxassetid://6023420473"
    local cornerBtn = Instance.new("UICorner")
    cornerBtn.CornerRadius = UDim.new(1, 0)
    cornerBtn.Parent = toggleButton
    toggleButton.Parent = screenGui

    local dragData = { dragging = false, startPos = nil, startMouse = nil }
    toggleButton.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragData.dragging = true
            dragData.startPos = toggleButton.Position
            dragData.startMouse = input.Position
        end
    end)
    toggleButton.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            dragData.dragging = false
        end
    end)
    UserInputService.InputChanged:Connect(function(input)
        if not dragData.dragging then return end
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseMovement then
            local delta = input.Position - dragData.startMouse
            toggleButton.Position = UDim2.new(
                dragData.startPos.X.Scale,
                dragData.startPos.X.Offset + delta.X,
                dragData.startPos.Y.Scale,
                dragData.startPos.Y.Offset + delta.Y
            )
        end
    end)

    local mainFrame = Instance.new("Frame")
    mainFrame.Size = UDim2.new(0, 320, 0, 290)
    mainFrame.Position = UDim2.new(0.5, -160, 0.5, -145)
    mainFrame.BackgroundColor3 = Color3.fromRGB(20, 20, 30)
    mainFrame.BackgroundTransparency = 0.1
    mainFrame.BorderSizePixel = 2
    mainFrame.BorderColor3 = Color3.fromRGB(150, 0, 0)
    mainFrame.Visible = false
    mainFrame.Parent = screenGui

    local title = Instance.new("TextLabel")
    title.Size = UDim2.new(1, 0, 0, 35)
    title.BackgroundTransparency = 1
    title.Text = "☠ РАЙОН НАСИЛИЯ ☠"
    title.TextColor3 = Color3.fromRGB(200, 0, 0)
    title.Font = Enum.Font.SourceSansBold
    title.TextSize = 22
    title.Parent = mainFrame

    local crosshairToggle = Instance.new("TextButton")
    crosshairToggle.Size = UDim2.new(0.8, 0, 0, 35)
    crosshairToggle.Position = UDim2.new(0.1, 0, 0.15, 0)
    crosshairToggle.BackgroundColor3 = Color3.fromRGB(70, 70, 70)
    crosshairToggle.Text = "🎯 CROSSHAIR: OFF"
    crosshairToggle.TextColor3 = Color3.new(1,1,1)
    crosshairToggle.Font = Enum.Font.SourceSansBold
    crosshairToggle.TextSize = 16
    crosshairToggle.Parent = mainFrame

    local fullbrightToggle = Instance.new("TextButton")
    fullbrightToggle.Size = UDim2.new(0.8, 0, 0, 35)
    fullbrightToggle.Position = UDim2.new(0.1, 0, 0.30, 0)
    fullbrightToggle.BackgroundColor3 = Color3.fromRGB(70, 70, 70)
    fullbrightToggle.Text = "☀️ FULLBRIGHT: OFF"
    fullbrightToggle.TextColor3 = Color3.new(1,1,1)
    fullbrightToggle.Font = Enum.Font.SourceSansBold
    fullbrightToggle.TextSize = 16
    fullbrightToggle.Parent = mainFrame

    local espToggle = Instance.new("TextButton")
    espToggle.Size = UDim2.new(0.8, 0, 0, 35)
    espToggle.Position = UDim2.new(0.1, 0, 0.45, 0)
    espToggle.BackgroundColor3 = Color3.fromRGB(70, 70, 70)
    espToggle.Text = "👁️ ESP: OFF"
    espToggle.TextColor3 = Color3.new(1,1,1)
    espToggle.Font = Enum.Font.SourceSansBold
    espToggle.TextSize = 16
    espToggle.Parent = mainFrame

    local closeBtn = Instance.new("TextButton")
    closeBtn.Size = UDim2.new(0.3, 0, 0, 35)
    closeBtn.Position = UDim2.new(0.35, 0, 0.70, 0)
    closeBtn.BackgroundColor3 = Color3.fromRGB(150, 0, 0)
    closeBtn.Text = "✖ ЗАКРЫТЬ"
    closeBtn.TextColor3 = Color3.new(1,1,1)
    closeBtn.Font = Enum.Font.SourceSansBold
    closeBtn.TextSize = 16
    closeBtn.Parent = mainFrame

    local crosshairGui = Instance.new("ScreenGui")
    crosshairGui.Name = "CrosshairGui"
    crosshairGui.ResetOnSpawn = false
    crosshairGui.IgnoreGuiInset = true
    crosshairGui.Parent = LocalPlayer.PlayerGui

    local crosshairDot = Instance.new("Frame")
    crosshairDot.Size = UDim2.new(0, 10, 0, 10)
    crosshairDot.AnchorPoint = Vector2.new(0.5, 0.5)
    crosshairDot.Position = UDim2.new(0.5, 0, 0.5, 0)
    crosshairDot.BackgroundColor3 = Color3.fromRGB(0, 255, 0)
    crosshairDot.BorderSizePixel = 0
    crosshairDot.Visible = false
    crosshairDot.Parent = crosshairGui

    local dotCorner = Instance.new("UICorner")
    dotCorner.CornerRadius = UDim.new(1, 0)
    dotCorner.Parent = crosshairDot

    toggleButton.Activated:Connect(function()
        menuOpen = not menuOpen
        mainFrame.Visible = menuOpen
    end)

    crosshairToggle.Activated:Connect(function()
        crosshairOn = not crosshairOn
        crosshairToggle.Text = crosshairOn and "🎯 CROSSHAIR: ON" or "🎯 CROSSHAIR: OFF"
        crosshairToggle.BackgroundColor3 = crosshairOn and Color3.fromRGB(0, 150, 0) or Color3.fromRGB(70, 70, 70)
        crosshairDot.Visible = crosshairOn
    end)

    fullbrightToggle.Activated:Connect(function()
        fullbrightOn = not fullbrightOn
        fullbrightToggle.Text = fullbrightOn and "☀️ FULLBRIGHT: ON" or "☀️ FULLBRIGHT: OFF"
        fullbrightToggle.BackgroundColor3 = fullbrightOn and Color3.fromRGB(0, 150, 0) or Color3.fromRGB(70, 70, 70)
        
        if fullbrightOn then
            oldBrightness = Lighting.Brightness
            oldAmbient = Lighting.Ambient
            Lighting.Brightness = 10
            Lighting.Ambient = Color3.fromRGB(255, 255, 255)
        else
            Lighting.Brightness = oldBrightness
            Lighting.Ambient = oldAmbient
        end
    end)

    espToggle.Activated:Connect(function()
        espOn = not espOn
        espToggle.Text = espOn and "👁️ ESP: ON" or "👁️ ESP: OFF"
        espToggle.BackgroundColor3 = espOn and Color3.fromRGB(0, 150, 0) or Color3.fromRGB(70, 70, 70)
        applyESP()
    end)

    closeBtn.Activated:Connect(function()
        mainFrame.Visible = false
        menuOpen = false
    end)

    local windowDrag = { dragging = false, startPos = nil, startMouse = nil }

    title.InputBegan:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            windowDrag.dragging = true
            windowDrag.startPos = mainFrame.Position
            windowDrag.startMouse = input.Position
        end
    end)

    title.InputEnded:Connect(function(input)
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseButton1 then
            windowDrag.dragging = false
        end
    end)

    title.InputChanged:Connect(function(input)
        if not windowDrag.dragging then return end
        if input.UserInputType == Enum.UserInputType.Touch or input.UserInputType == Enum.UserInputType.MouseMovement then
            local delta = input.Position - windowDrag.startMouse
            mainFrame.Position = UDim2.new(
                windowDrag.startPos.X.Scale,
                windowDrag.startPos.X.Offset + delta.X,
                windowDrag.startPos.Y.Scale,
                windowDrag.startPos.Y.Offset + delta.Y
            )
        end
    end)

    return screenGui
end

-- ============================================================
--  ЗАПУСК GUI
-- ============================================================
local gui = createGUI()

-- ============================================================
--  ФУНКЦИЯ ВОССТАНОВЛЕНИЯ
-- ============================================================
local function restoreGUI()
    local guiExists = LocalPlayer.PlayerGui:FindFirstChild("DistrictOfViolence")
    if not guiExists then
        gui = createGUI()
        if espOn then applyESP() end
        if fullbrightOn then
            Lighting.Brightness = 10
            Lighting.Ambient = Color3.fromRGB(255, 255, 255)
        end
        if crosshairOn then
            local crosshairGui = LocalPlayer.PlayerGui:FindFirstChild("CrosshairGui")
            if crosshairGui then
                local dot = crosshairGui:FindFirstChildWhichIsA("Frame")
                if dot then dot.Visible = true end
            end
        end
    end
end

-- ============================================================
--  АВТОВОССТАНОВЛЕНИЕ (при смерти, загрузке, изменении PlayerGui)
-- ============================================================
LocalPlayer.CharacterAdded:Connect(function()
    task.wait(0.5)
    restoreGUI()
end)

-- ДОПОЛНИТЕЛЬНОЕ ВОССТАНОВЛЕНИЕ: при любых изменениях PlayerGui
LocalPlayer.PlayerGui.ChildAdded:Connect(function()
    task.wait(0.1)
    restoreGUI()
end)
LocalPlayer.PlayerGui.ChildRemoved:Connect(function()
    task.wait(0.1)
    restoreGUI()
end)

local RunService = game:GetService("RunService")
RunService.Heartbeat:Connect(function()
    if not LocalPlayer then return end
    restoreGUI()
end)

-- ============================================================
--  АВТООБНОВЛЕНИЕ ESP
-- ============================================================
Players.PlayerAdded:Connect(function()
    task.wait(1)
    if espOn then applyESP() end
end)

Players.PlayerRemoving:Connect(function()
    task.wait(0.5)
    if espOn then applyESP() end
end)

workspace.DescendantAdded:Connect(function(obj)
    if espOn and obj:IsA("BasePart") then
        local name = obj.Name:lower()
        if name:find("generator") or name:find("gen") or name:find("генератор") or name:find("reactor") or name:find("engine") then
            task.wait(0.5)
            applyESP()
        end
    end
end)
